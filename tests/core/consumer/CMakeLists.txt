cmake_minimum_required(VERSION 3.24)
project(fgmp_external_consumer LANGUAGES CXX)
include(CTest)
if(NOT FGMP_SOURCE_DIR)
  message(FATAL_ERROR "Требуется каталог исходников Platform")
endif()
set(FGMC_BUILD_TESTS OFF CACHE BOOL "Тесты Core выключены потребителем" FORCE)
add_subdirectory("${FGMP_SOURCE_DIR}/core" core)
if(NOT TARGET fgm::cpu OR TARGET fgm_cpu_test OR NOT BUILD_TESTING)
  message(FATAL_ERROR "Нарушен контракт локального потребителя Core")
endif()
add_executable(consumer "${FGMP_SOURCE_DIR}/tests/core/sample.cpp")
target_link_libraries(consumer PRIVATE fgm::cpu)
add_test(NAME consumer-public-api COMMAND consumer "${FGMP_SOURCE_DIR}/core/tests/fixtures/c04-panel.fgm.json")
